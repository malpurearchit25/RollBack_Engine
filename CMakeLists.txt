cmake_minimum_required(VERSION 3.16)
project(RollbackEngine VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(rollback_core
    src/Entity.cpp src/GameStateSnapshot.cpp src/MoveCommand.cpp
    src/AttackCommand.cpp src/NoOpCommand.cpp src/StateBuffer.cpp
    src/StateDesyncException.cpp src/CommandFactory.cpp
    src/RollbackManager.cpp src/SimulationEngine.cpp)
target_include_directories(rollback_core PUBLIC include)
if(MSVC)
    target_compile_options(rollback_core PRIVATE /W4)
else()
    target_compile_options(rollback_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(rollback_engine src/engine_test.cpp)
target_link_libraries(rollback_engine PRIVATE rollback_core)
add_executable(server src/server_main.cpp)
target_link_libraries(server PRIVATE rollback_core Threads::Threads)
if(WIN32)
    target_compile_definitions(server PRIVATE _WIN32_WINNT=0x0A00)
    target_link_libraries(server PRIVATE ws2_32)
endif()
file(COPY web DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
include(CTest)
if(BUILD_TESTING)
    add_executable(rollback_tests tests/rollback_tests.cpp)
    target_link_libraries(rollback_tests PRIVATE rollback_core)
    add_test(NAME rollback_regressions COMMAND rollback_tests)
endif()
